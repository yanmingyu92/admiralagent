derive <- function(adsl, sources) {
  # derive AGEGR1: pooled age group as text, mapped from the numeric AGEGR1N
  # according to the controlled terminology:
  #   1 -> "<65", 2 -> "65-80", 3 -> ">80"
  # AGEGR1N is assumed to be derived earlier in ADSL (per the spec, AGEGR1 is
  # derived from ADSL.AGEGR1N).
  adsl |>
    mutate(
      AGEGR1 = case_when(
        AGEGR1N == 1 ~ "<65",
        AGEGR1N == 2 ~ "65-80",
        AGEGR1N == 3 ~ ">80",
        TRUE         ~ NA_character_
      )
    )
}

derive <- function(adsl, sources) {
  # derive TRT01PN: numeric code for TRT01P corresponding to randomized dose.
  # TRT01P is set from the planned arm (ARM) in the pilot study; map the
  # Xanomeline dose levels and Placebo to the numeric codes in the CT.
  adsl <- adsl |>
    mutate(
      TRT01PN = case_when(
        TRT01P == "Placebo" ~ 0L,
        TRT01P == "Xanomeline Low Dose" ~ 54L,
        TRT01P == "Xanomeline High Dose" ~ 81L,
        TRUE ~ NA_integer_
      )
    )
  adsl
}

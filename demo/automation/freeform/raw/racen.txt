derive <- function(adsl, sources) {
  # derive RACEN: numeric code for RACE using CDISC controlled terminology
  # (1=WHITE, 2=BLACK OR AFRICAN AMERICAN, 5=NATIVE HAWAIIAN OR OTHER PACIFIC
  #  ISLANDER, 6=AMERICAN INDIAN OR ALASKA NATIVE, 7=ASIAN)
  adsl <- adsl |>
    mutate(
      RACEN = case_when(
        toupper(trimws(RACE)) == "WHITE" ~ 1L,
        toupper(trimws(RACE)) == "BLACK OR AFRICAN AMERICAN" ~ 2L,
        toupper(trimws(RACE)) == "NATIVE HAWAIIAN OR OTHER PACIFIC ISLANDER" ~ 5L,
        toupper(trimws(RACE)) == "AMERICAN INDIAN OR ALASKA NATIVE" ~ 6L,
        toupper(trimws(RACE)) == "ASIAN" ~ 7L,
        TRUE ~ NA_integer_
      )
    )
  adsl
}

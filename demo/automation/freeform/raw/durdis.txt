derive <- function(adsl, sources) {
  # derive DURDIS: Duration of Disease (Months) = months between the first-visit
  # date (VISIT1DT) and disease onset date (DISONSET). Both VISIT1DT and DISONSET
  # are ADSL variables derived in earlier derivation steps; DURDIS is therefore a
  # simple arithmetic difference expressed in months.
  #
  # Months are computed as exact calendar months (whole months + fractional
  # remainder based on a 30.4375-day average month length) using the interval
  # between the two dates. Only computed when both dates are non-missing.

  adsl <- adsl |>
    mutate(
      DURDIS = if_else(
        !is.na(VISIT1DT) & !is.na(DISONSET),
        # number of whole calendar months plus the fractional remainder measured
        # against a standard 30.4375-day month
        as.numeric(
          (as.integer(format(VISIT1DT, "%Y")) - as.integer(format(DISONSET, "%Y"))) * 12 +
            (as.integer(format(VISIT1DT, "%m")) - as.integer(format(DISONSET, "%m")))
        ) +
          (as.integer(format(VISIT1DT, "%d")) - as.integer(format(DISONSET, "%d"))) / 30.4375,
        NA_real_
      )
    )

  adsl
}

derive <- function(adsl, sources) {
  # derive ETHNIC: carry DM.ETHNIC straight through into ADSL (a direct copy from
  # the SDTM DM domain on the same subject level, so no transformation is needed).
  # DM is the source dataset ADSL was built from, so ETHNIC already exists; we
  # still express this explicitly (merge/rename) in case ADSL did not carry it.
  adsl <- adsl |>
    derive_vars_merged(
      dataset_add = sources$dm,
      by_vars = vars(USUBJID),
      new_vars = vars(ETHNIC = ETHNIC)
    )

  adsl
}

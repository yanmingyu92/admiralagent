derive <- function(adsl, sources) {
  # TRT01AN: Numeric code for TRT01A, corresponding to the randomized dose.
  # The numeric treatment codes follow the study's controlled terminology:
  #   0  = Placebo
  #   54 = Xanomeline Low Dose
  #   81 = Xanomeline High Dose
  # TRT01A (actual treatment for period 01) is derived earlier in ADSL, so we
  # simply map its text values onto their numeric codes.
  adsl <- adsl |>
    mutate(
      TRT01AN = case_when(
        TRT01A == "Placebo"              ~ 0,
        TRT01A == "Xanomeline Low Dose"  ~ 54,
        TRT01A == "Xanomeline High Dose" ~ 81,
        is.na(TRT01A)                    ~ NA_real_,
        TRUE                             ~ NA_real_
      ),
      TRT01AN = as.integer(TRT01AN)
    )
  adsl
}

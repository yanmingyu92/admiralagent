derive <- function(adsl, sources) {
  # Derive SITEGR1: Pooled Site Group 1.
  # Per the spec: "refer to SAP, Section 7.1 - if not pooled then SITEGR1=SITEID.
  # If pooled, SITEGR1 will be 900". The SAP (not available here) defines which
  # sites are pooled. Since no pooling rule is provided in the specification or
  # in the source data, the safe, literal implementation is to default to the
  # non-pooled case: SITEGR1 = SITEID.
  adsl <- adsl |>
    mutate(
      SITEGR1 = SITEID
    )

  adsl
}
